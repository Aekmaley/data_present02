# ==============================================================================
# DATA CLEANING & REPRODUCIBLE PIPELINE (Python / Pandas)
# รหัสวิชา: 21906-2008 พื้นฐานการวิเคราะห์ข้อมูล / สมรรถนะ: ICT-ZXBP-381B
# ==============================================================================
import pandas as pd
import numpy as np

def clean_ecommerce_dataset(input_csv="raw_data.csv", output_csv="cleaned_data.csv"):
    # 1. Ingestion
    df = pd.read_csv(input_csv)
    
    # 2. De-duplication on Primary Key
    df = df.drop_duplicates(subset=['Transaction_ID'], keep='first').copy()
    
    # 3. Timestamp Repair (Fix Unix 2038 Bug & 2001 RTC Reset)
    df.loc[df['Transaction_ID'] == 'TX-10238', 'Timestamp'] = '2026-09-25 15:30:00'
    df.loc[df['Transaction_ID'] == 'TX-10258', 'Timestamp'] = '2026-09-27 04:10:00'
    df['Timestamp'] = pd.to_datetime(df['Timestamp'])
    
    # 4. Impute Missing Values
    df['platform'] = df['platform'].fillna('Unspecified').replace('', 'Unspecified')
    missing_price = df['Unit_Price_THB'].isna() | (df['Unit_Price_THB'] == 0)
    df.loc[missing_price, 'Unit_Price_THB'] = df['Transaction_Revenue_THB'] / df['Order_Qty']
    
    # 5. Outlier & Anomaly Corrections
    df.loc[df['Transaction_ID'] == 'TX-10038', 'Order_Qty'] = 4  # Fix negative typo
    df.loc[df['Unit_Price_THB'] > 100000, 'Unit_Price_THB'] = df['Transaction_Revenue_THB'] / df['Order_Qty']
    df.loc[df['Customer_Review_Rating'] > 5.0, 'Customer_Review_Rating'] = 4.8
    df.loc[df['Customer_Review_Rating'] < 1.0, 'Customer_Review_Rating'] = 2.0
    
    # 6. Mathematical Revenue Reconciliation
    df['Transaction_Revenue_THB'] = (df['Order_Qty'] * df['Unit_Price_THB']).round(2)
    
    # 7. Category Harmonization
    category_map = {
        'iPhone 15 Pro Max 256GB': 'Smartphones & Tablets',
        'Samsung Galaxy S24 Ultra': 'Smartphones & Tablets',
        'iPad Air 6 M2 WiFi': 'Smartphones & Tablets',
        'PlayStation 5 Slim 1TB': 'Gaming & Consoles',
        'Nintendo Switch OLED Model': 'Gaming & Consoles',
        'Logitech MX Master 3S Mouse': 'Computer Peripherals',
        'Keychron K2 V2 Mechanical Keyboard': 'Computer Peripherals',
        'Sony WH-1000XM5 ANC Headphone': 'Audio & Headphones',
        'AirPods Pro Gen 2 USB-C': 'Audio & Headphones',
        'DJI Osmo Pocket 3 Creator Combo': 'Cameras & Photography'
    }
    df['Harmonized_Category'] = df['Raw_Product_Name'].map(category_map).fillna('Other')
    
    df.to_csv(output_csv, index=False)
    return df

ใช้แถบ Filter ด้านบนเพื่อสลับดูตามแพลตฟอร์ม, หมวดหมู่, โซนคลัง หรือสถานะสินค้า กราฟและตัวเลขทั้งหมดจะปรับเปลี่ยนแบบเรียลไทม์ทันที (< 20ms)

2. การตรวจสอบคุณภาพ (Audit)

สลับไปที่แท็บ "Quality Audit" เพื่อดูตารางเปรียบเทียบ Before vs After และบันทึกการซ่อมแซมจุดบกพร่องทั้ง 13 รายการอย่างโปร่งใส

3. ข้อเสนอแนะเชิงธุรกิจ

ศึกษาข้อค้นพบในแท็บ "Insights & Strategy" ที่เรียบเรียงในรูปแบบ What → So What → Now What สำหรับนำไปประกอบการตัดสินใจจริง

4. การส่งออกและนำเสนอ

คลิกปุ่ม "ส่งออก CSV" เพื่อนำข้อมูล Cleaned Dataset ไปต่อยอด หรือคลิก "พิมพ์รายงาน (PDF)" เพื่อสร้างเอกสารส่งมอบกรรมการได้ทันที

ระบบวิเคราะห์ข้อมูลธุรกรรมการค้าพาณิชย์อิเล็กทรอนิกส์ & แดชบอร์ดนำเสนอผลงานการแข่งขันทักษะวิชาชีพ ทักษะการวิเคราะห์ข้อมูล ระดับ ปวช. ปีการศึกษา 2568 - 2572
